+import { render } from "@ember/test-helpers";
+import { module, test } from "qunit";
+import { setupRenderingTest } from "discourse/tests/helpers/component-test";
+import ISBFProfileSummary from "../../discourse/components/isbf-profile-summary";
+
+function outletArgs(canSeeActivity) {
+  const topics = Array.from({ length: 6 }, (_, index) => ({
+    id: index + 1,
+    fancyTitle: `Summary topic ${index + 1}`,
+    url: `/t/summary-topic/${index + 1}`,
+    like_count: 0,
+    created_at: "2026-10-20T01:00:00Z",
+  }));
+
+  return {
+    user: { id: 7, username: "charlie" },
+    model: {
+      topics,
+      replies: [],
+      most_replied_to_users: [],
+      most_liked_by_users: [],
+      most_liked_users: [],
+      can_see_user_actions: canSeeActivity,
+    },
+  };
+}
+
+module("ISBF | Integration | Component | ISBFProfileSummary", function (hooks) {
+  setupRenderingTest(hooks);
+
+  test("links a full list to the activity page while activity is visible", async function (assert) {
+    const args = outletArgs(true);
+
+    await render(
+      <template><ISBFProfileSummary @outletArgs={{args}} /></template>
+    );
+
+    assert
+      .dom(".isbf-profile-summary__more")
+      .exists({ count: 1 }, "the topics list links to more topics");
+  });
+
+  test("omits the activity link when activity is hidden", async function (assert) {
+    const args = outletArgs(false);
+
+    await render(
+      <template><ISBFProfileSummary @outletArgs={{args}} /></template>
+    );
+
+    assert
+      .dom(".isbf-profile-summary__toggle")
+      .exists("the list can still be expanded");
+    assert
+      .dom(".isbf-profile-summary__more")
+      .doesNotExist("no link leads into the hidden activity");
+  });
+});
